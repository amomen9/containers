``` Install Docker Desktop alongside Docker Engine


#list images/containers
docker images 	# all top-level
docker images -a	# all including intermediate
docker images nginx	# view images from a specific repository
docker images nginx:tagname	# list tag-specific images
docker images --digests [REPOSITORY][:TAG]


# remove docker image:
docker images
docker rmi [IMAGE_ID_OR_NAME]

## list containers
# List Running Containers:
docker ps
# List All Containers (Running and Stopped):
docker ps -a		# = docker container ls -a
# List Containers by ID (Quiet Mode):
docker ps -aq
# List the Latest Created Container:
docker ps -l
# List n Last Created Containers:
# Replace [NUMBER] with the number of containers you wish to list.
docker ps -n [NUMBER]
# List Containers with Size Information:
docker ps -s


# remove container
# stop it if running or force removal with -f flag
docker stop sql_container
docker rm sql_container


# stop/start containers
docker start/stop container_name


# list layers
docker history [IMAGE_NAME]
# remove layers: i found nothing


# docker run <image name> always downloads the image if not exists, creates the first container, and runs it.
# but if the image exists, it creates another additional container which might
# not be desirable and thus we should bear that in mind.
# create new container and exit and run no command
docker run <image name>
# create new container and run it interactively (remain in container) because of specifying bash command
docker run -it <image name> bash	# or /bin/bash
# create container with a command and exit
docker run -it  ubuntu bash -c lslogins
docker run -it  ubuntu lslogins
# create container with arbitrary name:
docker run -it --name my_container_name my_image /bin/bash


# docker exec
# The docker exec command is used to run a specific command in an already running container.
# It's a powerful feature that allows you to interact with or modify the container's environment in real-time.
docker exec [OPTIONS] CONTAINER COMMAND [ARG...]
docker exec -u root -it  sql_container_2
# options
{
• -d, --detach: Run the command in the background.
• -i, --interactive: Keep STDIN open even if not attached.
• -t, --tty: Allocate a pseudo-TTY, which simulates a terminal, like ssh or shell.
• --user: Username or UID to use when running the command.
• --env: Set environment variables.
• --workdir: Working directory inside the container.
}



### Portainer, portainer agent
```
1. run one of the following
- sanctioned
	# unsanctioned machine:
	one of the following 
		- docker pull portainer/portainer-ce:latest	# Community Edition
		- docker pull portainer/portainer-ee:latest	# Business Edition
	docker save -o portainer-portainer-ce:latest.tar portainer/portainer-ce:latest
	mv portainer-portainer-ce:latest.tar /var/www/html/download/
	chown -R www-data /var/www/html/download/
 
	# sanctioned machine
	docker load -i portainer-portainer-ce-latest.tar
	
- not sanctioned
	docker pull portainer/portainer-ce:latest

2. 
docker volume create portainer_data
# this will create a docker volume in /var/lib/docker/volumes/

3. create portainer container instance
docker run -d -p 9000:9000 --name=portainer --restart=unless-stopped -v /var/run/docker.sock:/var/run/docker.sock -v portainer_data:/data portainer/portainer-ce
docker run -d -p 8000:8000 -p 9443:9443 --name portainer --restart=always -v /var/run/docker.sock:/var/run/docker.sock -v portainer_data:/data portainer/portainer-ee:latest
docker run -d -p 8000:8000 -p 9443:9443 --name portainer-ee --restart=always -v /var/run/docker.sock:/var/run/docker.sock -v portainer_be_data:/data portainer/portainer-ee:2.26.0

# install on port 80 and another volume
docker run -d -p 80:9000 --name=portainer --restart=unless-stopped -v /var/run/docker.sock:/var/run/docker.sock -v portainer_data:/data portainer/portainer-ce



4. check if container is created and running
docker ps
5. finally, connect to your portainer web service using a web browser
http://192.168.171.81:9000/
6. create admin pass

# Install portainer agent
7.

  -p 9001:9001 \
  --name portainer_agent \
  --restart=always \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v /var/lib/docker/volumes:/var/lib/docker/volumes \
  portainer/agent:2.19.4

``` docker inspect container json file
### manipulate docker startup type behavior
```
Startup types:
{
no: The container will not automatically restart.
•  on-failure[:max-retries]: The container will restart if it exits due to an error, which manifests as a non-zero exit code. Optionally, you can limit the number of restart attempts with max-retries.

•  always: The container will always restart unless it is manually stopped.

•  unless-stopped: The container will always restart unless it has been manually stopped.

}

``` install docker image offline sanction sanctions sanctioned
### Setup docker registry
```
# Run the Registry as a Docker Container:
docker run -d -p 5000:5000 --restart always --name registry registry:latest
# tag your image with the registry's address:
docker tag my-image localhost:5000/my-image
# push the image to registry
docker push localhost:5000/my-image
# Persisting Data:
docker run -d -p 5000:5000 --restart always --name registry -v /path/on/host:/var/lib/registry registry:2

When you forcibly remove a Docker image using the docker rmi -f command, 
Docker does not automatically stop and remove the running containers associated 
with that image. Instead, it un-tags the image, and the containers continue to run 
but are now associated with the image ID rather than the image name
Here's what happens in detail:
1. Image Un-tagging: The image is un-tagged, meaning its name and 
 tag are removed, but the image data remains on the system


2. Containers Continue Running: Any running containers that
 were using the image will continue to run, but their IMAGE
 field will now show the image ID instead of the name.
 
  
3. Image Removal: The actual image data is not removed
 until all containers using that image are stopped and removed.
